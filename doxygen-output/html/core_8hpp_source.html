235 static auto format(const Value& value, format_context& ctx) {
236 if (value.flag() == MatchFlags::EMPTY) {
237 return format_to(ctx.out(), "<empty>");
238 }
239 if (value.flag() == MatchFlags::STRING) {
240 if (auto text = value.asString()) {
241 return format_to(ctx.out(), "\"{}\"", *text);
242 }
243 return format_to(ctx.out(), "\"\"");
244 }
245 format_to(ctx.out(), "[");
246 for (std::size_t i = 0; i < value.size(); ++i) {
247 if (i != 0U) {
248 format_to(ctx.out(), " ");
249 }
250 format_to(ctx.out(), "{:02x}", value.data()[i]);
251 }
252 return format_to(ctx.out(), "]");
253 }
